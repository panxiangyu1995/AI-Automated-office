# VSCode 编辑器注册与解析实现详解

## 一、编辑器注册机制

### 1.1 注册入口

VSCode 提供了多种编辑器注册方式：

#### 1.1.1 通过扩展点注册（声明式）

在 `package.json` 中声明：

```json
{
    "contributes": {
        "customEditors": [{
            "viewType": "myEditor.viewType",
            "displayName": "My Editor",
            "priority": "default",
            "selector": [{ "filenamePattern": "*.myext" }]
        }]
    }
}
```

**处理流程**：

1. 扩展加载时，`customEditorsExtensionPoint` 被触发
2. `ContributedCustomEditors` 类收集所有贡献的编辑器
3. `CustomEditorService` 将贡献转换为编辑器注册

**源码位置**：`src/vs/workbench/contrib/customEditor/common/extensionPoint.ts`

```typescript
export const customEditorsExtensionPoint = ExtensionsRegistry.registerExtensionPoint<ICustomEditorsExtensionPoint[]>({
    extensionPoint: 'customEditors',
    deps: [languagesExtPoint],
    jsonSchema: { ... },
    activationEventsGenerator: function* (contribs: readonly ICustomEditorsExtensionPoint[]) {
        for (const contrib of contribs) {
            const viewType = contrib[Fields.viewType];
            if (viewType) {
                yield `onCustomEditor:${viewType}`;
            }
        }
    },
});
```

#### 1.1.2 通过服务注册（编程式）

使用 `IEditorResolverService.registerEditor()` 方法：

```typescript
const disposable = editorResolverService.registerEditor(
    '*.myext',                    // Glob 模式
    {
        id: 'myEditor.viewType',  // 编辑器 ID
        label: 'My Editor',       // 显示标签
        detail: 'My Extension',   // 提供者名称
        priority: RegisteredEditorPriority.default  // 优先级
    },
    {
        singlePerResource: true,  // 每个资源只能打开一个实例
        canSupportResource: (resource) => true  // 资源支持检查
    },
    {
        createEditorInput: (editorInput, group) => {
            return { editor: myEditorInput, options: {} };
        },
        createDiffEditorInput: (diffInput, group) => {
            return { editor: myDiffEditorInput, options: {} };
        }
    }
);
```

### 1.3 编辑器存储结构

编辑器注册信息存储在嵌套的 Map 结构中：

```typescript
// 外层 Map：Glob 模式 -> 内层 Map
// 内层 Map：编辑器 ID -> 编辑器数组
private _editors: Map<string | glob.IRelativePattern, Map<string, RegisteredEditors>> = new Map();

// 扁平化后的编辑器列表
private _flattenedEditors: Map<string | glob.IRelativePattern, RegisteredEditors> = new Map();
```

**扁平化处理**：

```typescript
private _flattenEditorsMap() {
    if (!this._shouldReFlattenEditors) {
        return this._flattenedEditors;
    }
    this._shouldReFlattenEditors = false;
    
    const editors = new Map<string | glob.IRelativePattern, RegisteredEditors>();
    for (const [glob, value] of this._editors) {
        const registeredEditors: RegisteredEditors = [];
        for (const editors of value.values()) {
            let registeredEditor: RegisteredEditor | undefined = undefined;
            for (const editor of editors) {
                if (!registeredEditor) {
                    registeredEditor = {
                        editorInfo: editor.editorInfo,
                        globPattern: editor.globPattern,
                        options: {},
                        editorFactoryObject: {}
                    };
                }
                // 合并选项和工厂函数
                registeredEditor.options = { ...registeredEditor.options, ...editor.options };
                registeredEditor.editorFactoryObject = { 
                    ...registeredEditor.editorFactoryObject, 
                    ...editor.editorFactoryObject 
                };
            }
            if (registeredEditor) {
                registeredEditors.push(registeredEditor);
            }
        }
        editors.set(glob, registeredEditors);
    }
    return editors;
}
```

## 二、编辑器解析流程

### 2.1 主解析入口

```typescript
async resolveEditor(editor: IUntypedEditorInput, preferredGroup: PreferredGroup | undefined): Promise<ResolvedEditor> {
    // 1. 更新扁平化编辑器列表
    this._flattenedEditors = this._flattenEditorsMap();

    // 2. 处理并排编辑器
    if (isResourceSideBySideEditorInput(editor)) {
        return this.doResolveSideBySideEditor(editor, preferredGroup);
    }

    // 3. 解析编辑器和组
    const resolvedUntypedAndGroup = await this.resolveUntypedInputAndGroup(editor, preferredGroup);
    if (!resolvedUntypedAndGroup) {
        return ResolvedStatus.NONE;
    }
    const [untypedEditor, group, activation] = resolvedUntypedAndGroup;

    // 4. 获取资源
    let resource = EditorResourceAccessor.getCanonicalUri(untypedEditor, { supportSideBySide: SideBySideEditor.PRIMARY });

    // 5. 等待扩展激活（如果需要）
    if (this.cache && resource && this.resourceMatchesCache(resource)) {
        await this.extensionService.whenInstalledExtensionsRegistered();
    }

    // 6. 处理未定义资源
    if (resource === undefined) {
        resource = URI.from({ scheme: Schemas.untitled });
    } else if (resource.scheme === undefined || resource === null) {
        return ResolvedStatus.NONE;
    }

    // 7. 处理用户选择
    if (untypedEditor.options?.override === EditorResolution.PICK) {
        const picked = await this.doPickEditor(untypedEditor);
        if (!picked) {
            return ResolvedStatus.ABORT;
        }
        untypedEditor.options = picked;
    }

    // 8. 获取编辑器
    let { editor: selectedEditor, conflictingDefault } = this.getEditor(resource, untypedEditor.options?.override);
    
    // 9. 处理未找到编辑器的情况
    if (!selectedEditor && (untypedEditor.options?.override || isEditorInputWithOptions(editor))) {
        return ResolvedStatus.NONE;
    } else if (!selectedEditor) {
        const resolvedEditor = this.getEditor(resource, DEFAULT_EDITOR_ASSOCIATION.id);
        selectedEditor = resolvedEditor?.editor;
        conflictingDefault = resolvedEditor?.conflictingDefault;
        if (!selectedEditor) {
            return ResolvedStatus.NONE;
        }
    }

    // 10. 处理差异编辑器
    if (isResourceDiffEditorInput(untypedEditor) && untypedEditor.options?.override === undefined) {
        // 检查两侧是否使用相同编辑器
        let resource2 = EditorResourceAccessor.getCanonicalUri(untypedEditor, { supportSideBySide: SideBySideEditor.SECONDARY });
        if (!resource2) {
            resource2 = URI.from({ scheme: Schemas.untitled });
        }
        const { editor: selectedEditor2 } = this.getEditor(resource2, undefined);
        if (!selectedEditor2 || selectedEditor.editorInfo.id !== selectedEditor2.editorInfo.id) {
            // 两侧编辑器不同，回退到默认编辑器
            const { editor: selectedDiff } = this.getEditor(resource, DEFAULT_EDITOR_ASSOCIATION.id);
            selectedEditor = selectedDiff;
        }
    }

    // 11. 创建编辑器输入
    const input = await this.doResolveEditor(untypedEditor, group, selectedEditor);
    
    // 12. 处理冲突默认值
    if (conflictingDefault && input) {
        await this.doHandleConflictingDefaults(resource, selectedEditor.editorInfo.label, untypedEditor, input.editor, group);
    }

    return input ? { ...input, group } : ResolvedStatus.ABORT;
}
```

### 2.2 编辑器选择逻辑

```typescript
private getEditor(resource: URI, editorId: string | EditorResolution.EXCLUSIVE_ONLY | undefined): 
    { editor: RegisteredEditor | undefined; conflictingDefault: boolean } {
    
    // 查找指定 ID 的编辑器
    const findMatchingEditor = (editors: RegisteredEditors, viewType: string) => {
        return editors.find((editor) => {
            if (editor.options?.canSupportResource !== undefined) {
                return editor.editorInfo.id === viewType && editor.options.canSupportResource(resource);
            }
            return editor.editorInfo.id === viewType;
        });
    };

    // 如果指定了编辑器 ID，直接查找
    if (editorId && editorId !== EditorResolution.EXCLUSIVE_ONLY) {
        const registeredEditors = this._registeredEditors;
        return {
            editor: findMatchingEditor(registeredEditors, editorId),
            conflictingDefault: false
        };
    }

    // 查找所有匹配的编辑器
    const editors = this.findMatchingEditors(resource);
    const associationsFromSetting = this.getAssociationsForResource(resource);
    
    // 确定最低优先级
    const minPriority = editorId === EditorResolution.EXCLUSIVE_ONLY 
        ? RegisteredEditorPriority.exclusive 
        : RegisteredEditorPriority.builtin;
    
    let possibleEditors = editors.filter(editor => 
        priorityToRank(editor.editorInfo.priority) >= priorityToRank(minPriority) 
        && editor.editorInfo.id !== DEFAULT_EDITOR_ASSOCIATION.id
    );
    
    if (possibleEditors.length === 0) {
        return {
            editor: associationsFromSetting[0] && minPriority !== RegisteredEditorPriority.exclusive 
                ? findMatchingEditor(editors, associationsFromSetting[0].viewType) 
                : undefined,
            conflictingDefault: false
        };
    }

    // 选择编辑器
    const selectedViewType = possibleEditors[0].editorInfo.priority === RegisteredEditorPriority.exclusive
        ? possibleEditors[0].editorInfo.id
        : associationsFromSetting[0]?.viewType
        || possibleEditors.find(editor => !editor.options?.canSupportResource || editor.options.canSupportResource(resource))?.editorInfo.id
        || possibleEditors[0].editorInfo.id;

    // 检查冲突
    let conflictingDefault = false;
    possibleEditors = possibleEditors
        .filter(editor => editor.editorInfo.priority !== RegisteredEditorPriority.exclusive)
        .filter(editor => !editor.options?.canSupportResource || editor.options.canSupportResource(resource));
    
    if (associationsFromSetting.length === 0 && possibleEditors.length > 1) {
        conflictingDefault = true;
    }

    return {
        editor: findMatchingEditor(editors, selectedViewType),
        conflictingDefault
    };
}
```

### 2.3 匹配编辑器查找

```typescript
private findMatchingEditors(resource: URI): RegisteredEditor[] {
    // 获取用户设置关联
    const userSettings = this.getAssociationsForResource(resource);
    const matchingEditors: RegisteredEditor[] = [];
    
    // 遍历所有已注册编辑器
    for (const [key, editors] of this._flattenedEditors) {
        for (const editor of editors) {
            const foundInSettings = userSettings.find(setting => setting.viewType === editor.editorInfo.id);
            
            // 匹配条件：
            // 1. 在用户设置中找到（且不是 exclusive）
            // 2. 或者 Glob 模式匹配资源
            if ((foundInSettings && editor.editorInfo.priority !== RegisteredEditorPriority.exclusive) 
                || globMatchesResource(key, resource)) {
                matchingEditors.push(editor);
            }
        }
    }
    
    // 按优先级排序
    return matchingEditors.sort((a, b) => {
        // 如果优先级相同，更长的 Glob 更具体
        if (priorityToRank(b.editorInfo.priority) === priorityToRank(a.editorInfo.priority) 
            && typeof b.globPattern === 'string' && typeof a.globPattern === 'string') {
            return b.globPattern.length - a.globPattern.length;
        }
        return priorityToRank(b.editorInfo.priority) - priorityToRank(a.editorInfo.priority);
    });
}
```

## 三、Glob 匹配实现

### 3.1 匹配函数

```typescript
function globMatchesResource(globPattern: string | glob.IRelativePattern, resource: URI): boolean {
    // 排除特定协议
    const excludedSchemes = new Set([
        Schemas.extension,
        Schemas.webviewPanel,
        Schemas.vscodeWorkspaceTrust,
        Schemas.vscodeSettings
    ]);
    
    if (excludedSchemes.has(resource.scheme)) {
        return false;
    }
    
    // 确定匹配目标
    const matchOnPath = typeof globPattern === 'string' && globPattern.indexOf(posix.sep) >= 0;
    const target = matchOnPath 
        ? `${resource.scheme}:${resource.path}`  // 路径匹配
        : basename(resource);                     // 文件名匹配
    
    return glob.match(globPattern, target, { ignoreCase: true });
}
```

### 3.2 匹配模式类型

| 模式类型 | 示例 | 匹配目标 |
|----------|------|----------|
| 文件名模式 | `*.txt` | 仅匹配文件名 |
| 路径模式 | `**/config/*.json` | 匹配完整路径 |
| 相对模式 | `{ base: '/workspace', pattern: '*.ts' }` | 相对于基础路径 |

## 四、编辑器输入创建

### 4.1 创建流程

```typescript
private async doResolveEditor(
    editor: IUntypedEditorInput, 
    group: IEditorGroup, 
    selectedEditor: RegisteredEditor
): Promise<EditorInputWithOptions | undefined> {
    let options = editor.options;
    const resource = EditorResourceAccessor.getCanonicalUri(editor, { supportSideBySide: SideBySideEditor.PRIMARY });
    
    // 设置激活选项
    if (options && typeof options.activation === 'undefined') {
        options = { ...options, activation: options.preserveFocus ? EditorActivation.RESTORE : undefined };
    }

    // 处理合并编辑器
    if (isResourceMergeEditorInput(editor)) {
        if (!selectedEditor.editorFactoryObject.createMergeEditorInput) return;
        const inputWithOptions = await selectedEditor.editorFactoryObject.createMergeEditorInput(editor, group);
        return { editor: inputWithOptions.editor, options: inputWithOptions.options ?? options };
    }

    // 处理差异编辑器
    if (isResourceDiffEditorInput(editor)) {
        if (!selectedEditor.editorFactoryObject.createDiffEditorInput) return;
        const inputWithOptions = await selectedEditor.editorFactoryObject.createDiffEditorInput(editor, group);
        return { editor: inputWithOptions.editor, options: inputWithOptions.options ?? options };
    }

    // 处理多差异编辑器
    if (isResourceMultiDiffEditorInput(editor)) {
        if (!selectedEditor.editorFactoryObject.createMultiDiffEditorInput) return;
        const inputWithOptions = await selectedEditor.editorFactoryObject.createMultiDiffEditorInput(editor, group);
        return { editor: inputWithOptions.editor, options: inputWithOptions.options ?? options };
    }

    // 处理未保存编辑器
    if (isUntitledResourceEditorInput(editor)) {
        if (!selectedEditor.editorFactoryObject.createUntitledEditorInput) return;
        const inputWithOptions = await selectedEditor.editorFactoryObject.createUntitledEditorInput(editor, group);
        return { editor: inputWithOptions.editor, options: inputWithOptions.options ?? options };
    }

    // 处理单资源编辑器
    const singleEditorPerResource = typeof selectedEditor.options?.singlePerResource === 'function' 
        ? selectedEditor.options.singlePerResource() 
        : selectedEditor.options?.singlePerResource;
    
    if (singleEditorPerResource) {
        const existingEditors = this.findExistingEditorsForResource(resource, selectedEditor.editorInfo.id);
        if (existingEditors.length) {
            const editor = await this.moveExistingEditorForResource(existingEditors, group);
            if (editor) {
                return { editor, options };
            }
            return;
        }
    }

    // 创建新编辑器输入
    if (!selectedEditor.editorFactoryObject.createEditorInput) return;
    const inputWithOptions = await selectedEditor.editorFactoryObject.createEditorInput(editor, group);
    options = inputWithOptions.options ?? options;
    
    return { editor: inputWithOptions.editor, options };
}
```

## 五、用户设置处理

### 5.1 获取用户关联

```typescript
getAllUserAssociations(): EditorAssociations {
    const inspectedEditorAssociations = this.configurationService.inspect<{ [fileNamePattern: string]: string }>(
        editorsAssociationsSettingId
    ) || {};
    
    const defaultAssociations = inspectedEditorAssociations.defaultValue ?? {};
    const workspaceAssociations = inspectedEditorAssociations.workspaceValue ?? {};
    const userAssociations = inspectedEditorAssociations.userValue ?? {};
    
    const rawAssociations: { [fileNamePattern: string]: string } = { ...workspaceAssociations };
    
    // 合并关联，工作区设置优先
    for (const [key, value] of Object.entries({ ...defaultAssociations, ...userAssociations })) {
        if (rawAssociations[key] === undefined) {
            rawAssociations[key] = value;
        }
    }
    
    // 转换为数组格式
    const associations = [];
    for (const [key, value] of Object.entries(rawAssociations)) {
        associations.push({
            filenamePattern: key,
            viewType: value
        });
    }
    return associations;
}
```

### 5.2 更新用户关联

```typescript
updateUserAssociations(globPattern: string, editorID: string): void {
    const newAssociation: EditorAssociation = { viewType: editorID, filenamePattern: globPattern };
    const currentAssociations = this.getAllUserAssociations();
    const newSettingObject = Object.create(null);
    
    for (const association of [...currentAssociations, newAssociation]) {
        if (association.filenamePattern) {
            newSettingObject[association.filenamePattern] = association.viewType;
        }
    }
    
    this.configurationService.updateValue(editorsAssociationsSettingId, newSettingObject);
}
```
