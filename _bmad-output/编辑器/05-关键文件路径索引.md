## 十一、目录结构概览

```
src/vs/workbench/
├── browser/parts/editor/          # 编辑器 UI 组件
│   ├── editor.contribution.ts     # 编辑器贡献注册
│   ├── editorPane.ts              # 编辑器面板基类
│   ├── textEditor.ts              # 文本编辑器
│   ├── textDiffEditor.ts          # 差异编辑器
│   └── sideBySideEditor.ts        # 并排编辑器
│
├── common/editor/                 # 编辑器通用模型
│   ├── editor.ts                  # 编辑器核心类型
│   ├── editorInput.ts             # 编辑器输入基类
│   └── diffEditorInput.ts         # 差异编辑器输入
│
├── services/editor/               # 编辑器服务
│   ├── common/
│   │   ├── editorResolverService.ts  # 解析服务接口
│   │   └── editorService.ts          # 编辑器服务接口
│   └── browser/
│       └── editorResolverService.ts  # 解析服务实现
│
├── contrib/customEditor/          # 自定义编辑器
│   ├── common/
│   │   ├── extensionPoint.ts      # 扩展点定义
│   │   └── customEditor.ts        # 类型定义
│   └── browser/
│       ├── customEditors.ts       # 自定义编辑器服务
│       └── customEditorInput.ts   # 编辑器输入
│
├── contrib/notebook/              # Notebook 编辑器
├── contrib/mergeEditor/           # 合并编辑器
├── contrib/searchEditor/          # 搜索编辑器
└── contrib/files/browser/editors/ # 文件编辑器

extensions/
├── media-preview/                 # 媒体预览扩展
├── ipynb/                         # Jupyter Notebook 扩展
└── markdown-language-features/    # Markdown 扩展
```
