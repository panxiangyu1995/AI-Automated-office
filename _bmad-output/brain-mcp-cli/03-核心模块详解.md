# brain-mcp-cli 核心模块详解

## 1. 数据摄入系统（Ingest System）

### 1.1 系统概述

数据摄入系统负责从多个AI对话平台导入对话历史，并将其标准化为统一的Parquet格式。

### 1.2 核心组件

#### 1.2.1 BaseIngester（基类）

**文件位置**：`brain_mcp/ingest/base.py`

**核心职责**：
- 定义摄入器的标准接口
- 提供增量摄入的默认实现

**接口定义**：
```python
class BaseIngester(ABC):
    @property
    @abstractmethod
    def source_type(self) -> str:
        """唯一标识符，如 'claude-code', 'chatgpt'"""
        ...
    
    @property
    @abstractmethod
    def display_name(self) -> str:
        """人类可读名称，如 'Claude Code', 'ChatGPT'"""
        ...
    
    @abstractmethod
    def discover(self) -> list[dict]:
        """
        发现数据源
        返回: [{"path": "...", "count_hint": N}, ...]
        """
        ...
    
    @abstractmethod
    def ingest(self, source_path: str) -> list[dict]:
        """
        摄入数据
        返回: 标准化的消息记录列表
        """
        ...
    
    def incremental_ingest(
        self, source_path: str, state: dict
    ) -> tuple[list[dict], dict]:
        """
        增量摄入（可选实现）
        返回: (新记录, 更新后的状态)
        """
        records = self.ingest(source_path)
        return records, {"last_full_ingest": datetime.now().isoformat()}
```

```python
def make_record(
    source: str,
    conversation_id: str,
    role: str,
    content: str,
    timestamp: datetime,
    msg_index: int = 0,
    *,
    model: str = None,
    project: str = None,
    conversation_title: str = None,
    message_id: str = None,
    parent_id: str = None,
    temporal_precision: str = "exact",
) -> dict:
    """
    创建符合Canonical Schema的消息记录
    自动计算：word_count, char_count, has_code, has_url, has_question
    自动处理：内容清理、UTF-8编码、长度限制
    """
    content = (content or "").strip()
    if not content:
        return None
    
    # UTF-8编码处理
    try:
        content.encode("utf-8")
    except UnicodeEncodeError:
        content = content.encode("utf-8", errors="replace").decode("utf-8")
    
    return {
        "source": source,
        "model": model,
        "project": project,
        "conversation_id": conversation_id,
        "conversation_title": conversation_title or "Untitled",
        "created": timestamp,
        "updated": timestamp,
        "year": timestamp.year,
        "month": timestamp.month,
        "day_of_week": timestamp.strftime("%A"),
        "hour": timestamp.hour,
        "message_id": message_id or f"{conversation_id}_{msg_index}",
        "parent_id": parent_id,
        "msg_index": msg_index,
        "msg_timestamp": timestamp,
        "timestamp_is_fallback": 0 if temporal_precision == "exact" else 1,
        "role": role,
        "content_type": "text",
        "is_first": 1 if msg_index == 0 else 0,
        "is_last": 0,
        "word_count": len(content.split()),
        "char_count": len(content),
        "conversation_msg_count": 0,
        "has_code": 1 if "```" in content or re.search(r"def |function |class ", content) else 0,
        "has_url": 1 if re.search(r"https?://", content) else 0,
        "has_question": 1 if "?" in content else 0,
        "has_attachment": 0,
        "has_citation": 0,
        "content": content[:50000],  # 限制50K字符
        "temporal_precision": temporal_precision,
    }

def finalize_conversation(records: list[dict]) -> list[dict]:
    """
    完善对话级别字段：
    - 设置 is_first 和 is_last
    - 设置 conversation_msg_count
    - 修正 msg_index
    """
    if not records:
        return records
    
    msg_count = len(records)
    for i, r in enumerate(records):
        r["conversation_msg_count"] = msg_count
        r["is_first"] = 1 if i == 0 else 0
        r["is_last"] = 1 if i == msg_count - 1 else 0
        r["msg_index"] = i
    
    return records
```

#### 1.2.3 Registry（注册表）

**文件位置**：`brain_mcp/ingest/registry.py`

**核心机制**：
```python
# 全局注册表
_REGISTRY: dict[str, BaseIngester] = {}

def register(cls):
    """
    类装饰器，自动注册摄入器
    
    使用方式：
        @register
        class MyIngester(BaseIngester):
            ...
    """
    instance = cls()
    _REGISTRY[instance.source_type] = instance
    return cls

def get_ingester(source_type: str) -> BaseIngester | None:
    """根据source_type获取摄入器实例"""
    return _REGISTRY.get(source_type)

def get_all_ingesters() -> dict[str, BaseIngester]:
    """获取所有已注册的摄入器"""
    return dict(_REGISTRY)

def discover_all() -> dict[str, list[dict]]:
    """
    运行所有摄入器的discover方法
    返回: {source_type: [发现的数据源列表]}
    """
    results = {}
    for name, ingester in _REGISTRY.items():
        try:
            found = ingester.discover()
            if found:
                results[name] = found
        except Exception as e:
            print(f"Discovery failed for {name}: {e}")
    return results
```

## 2. 嵌入系统（Embed System）

### 2.1 系统概述

嵌入系统负责将对话消息转换为向量表示，支持语义相似度搜索。

### 2.2 核心组件

#### 2.2.1 EmbeddingProvider（抽象基类）

**文件位置**：`brain_mcp/embed/provider.py`

**接口定义**：
```python
class EmbeddingProvider(ABC):
    @abstractmethod
    def embed_batch(self, texts: list[str]) -> list[list[float]]:
        """批量嵌入文本"""
        ...
    
    @abstractmethod
    def embed_query(self, text: str) -> list[float]:
        """嵌入单个查询（可能优化）"""
        ...
    
    @property
    @abstractmethod
    def dimension(self) -> int:
        """向量维度"""
        ...
    
    @property
    @abstractmethod
    def provider_name(self) -> str:
        """提供者名称"""
        ...
    
    @property
    @abstractmethod
    def model_name(self) -> str:
        """模型名称"""
        ...
```

#### 2.2.2 FastEmbedProvider（默认实现）

**特点**：
- 基于ONNX Runtime
- 无需PyTorch
- 轻量级（107MB）
- 默认模型：BAAI/bge-small-en-v1.5（384维）

**实现**：
```python
class FastEmbedProvider(EmbeddingProvider):
    DEFAULT_MODEL = "BAAI/bge-small-en-v1.5"
    
    def __init__(self, model_name: Optional[str] = None):
        from fastembed import TextEmbedding
        
        self._model_name = model_name or self.DEFAULT_MODEL
        self._model = TextEmbedding(model_name=self._model_name)
        
        # 通过测试嵌入确定维度
        test = list(self._model.embed(["test"]))
        self._dim = len(test[0])
    
    def embed_batch(self, texts: list[str]) -> list[list[float]]:
        if not texts:
            return []
        embeddings = list(self._model.embed(texts))
        return [emb.tolist() for emb in embeddings]
    
    def embed_query(self, text: str) -> list[float]:
        # 使用优化的query_embed
        results = list(self._model.query_embed(text))
        return results[0].tolist()
    
    @property
    def dimension(self) -> int:
        return self._dim
    
    @property
    def provider_name(self) -> str:
        return f"fastembed ({self._model_name}, {self._dim} dims)"
```

#### 2.2.3 SentenceTransformerProvider（备选实现）

**特点**：
- 基于PyTorch
- 较重（~1.3GB）
- 默认模型：nomic-ai/nomic-embed-text-v1.5（768维）
- 支持更多模型选择

**实现**：
```python
class SentenceTransformerProvider(EmbeddingProvider):
    DEFAULT_MODEL = "nomic-ai/nomic-embed-text-v1.5"
    
    def __init__(self, model_name: Optional[str] = None):
        from sentence_transformers import SentenceTransformer
        
        self._model_name = model_name or self.DEFAULT_MODEL
        self._model = SentenceTransformer(
            self._model_name, trust_remote_code=True
        )
        
        test = self._model.encode(["test"])
        self._dim = test.shape[1]
    
    def embed_batch(self, texts: list[str]) -> list[list[float]]:
        if not texts:
            return []
        
        # Nomic模型需要前缀
        if "nomic" in self._model_name.lower():
            texts = [f"search_document: {t}" for t in texts]
        
        embeddings = self._model.encode(texts, show_progress_bar=False)
        return [emb.tolist() for emb in embeddings]
    
    def embed_query(self, text: str) -> list[float]:
        if "nomic" in self._model_name.lower():
            text = f"search_query: {text}"
        
        emb = self._model.encode([text], show_progress_bar=False)[0]
        return emb.tolist()
```

#### 2.2.4 提供者自动检测

```python
def _detect_available_provider() -> str:
    """检测可用的嵌入后端"""
    # 优先尝试fastembed
    try:
        import fastembed
        return "fastembed"
    except ImportError:
        pass
    
    # 回退到sentence-transformers
    try:
        import sentence_transformers
        return "sentence-transformers"
    except ImportError:
        pass
    
    raise ImportError(
        "No embedding backend found. Install one of:\n"
        "  pip install 'brain-mcp[embed]'           # fastembed (推荐)\n"
        "  pip install 'brain-mcp[embed-torch]'     # sentence-transformers\n"
    )

def get_provider(
    model_name: Optional[str] = None,
    force_provider: Optional[str] = None,
) -> EmbeddingProvider:
    """获取缓存的嵌入提供者单例"""
    global _provider
    
    if _provider is None:
        provider_type = force_provider or _detect_available_provider()
        
        if provider_type == "fastembed":
            _provider = FastEmbedProvider(model_name)
        elif provider_type == "sentence-transformers":
            _provider = SentenceTransformerProvider(model_name)
        else:
            raise ValueError(f"Unknown provider: {provider_type}")
    
    return _provider
```

### 2.4 维度不匹配处理

```python
def _check_dimension_mismatch(db, provider):
    """检查现有向量维度是否与当前提供者匹配"""
    try:
        if "message" not in db.table_names():
            return False, None
        
        tbl = db.open_table("message")
        if tbl.count_rows() == 0:
            return False, None
        
        # 采样一行获取维度
        sample = tbl.to_pandas().head(1)
        if "embedding" in sample.columns:
            existing_emb = sample["embedding"].iloc[0]
            existing_dim = len(existing_emb)
            
            if existing_dim != provider.dimension:
                return True, existing_dim
    except Exception:
        pass
    
    return False, None

# 使用示例
mismatch, existing_dim = _check_dimension_mismatch(db, provider)
if mismatch:
    print(f"⚠️  维度不匹配: 现有{existing_dim}d, 当前{provider.dimension}d")
    print("   运行 brain-mcp embed --rebuild 重建")
    return

### 3.3 摘要生成流程

```python
def summarize_conversations(full: bool = False):
    """生成对话摘要"""
    
    # 1. 加载对话
    df = pd.read_parquet(parquet_path)
    
    # 2. 加载已有摘要（增量）
    existing_ids = set()
    if not full and summaries_jsonl.exists():
        with open(summaries_jsonl) as f:
            for line in f:
                obj = json.loads(line)
                existing_ids.add(obj["conversation_id"])
    
    # 3. 按对话分组
    conversations = df.groupby("conversation_id")
    
    # 4. 生成摘要
    for conv_id, messages in conversations:
        if conv_id in existing_ids:
            continue
        
        # 构建对话文本
        conv_text = "\n\n".join([
            f"{m['role']}: {m['content']}"
            for m in messages.sort_values("msg_index").to_dict("records")
        ])
        
        # 调用LLM
        prompt = SUMMARY_PROMPT.format(conversation=conv_text)
        response = call_llm(prompt)
        
        # 解析JSON
        summary_data = json.loads(response)
        
        # 添加元数据
        summary_data["conversation_id"] = conv_id
        summary_data["summarized_at"] = datetime.now().isoformat()
        summary_data["msg_count"] = len(messages)
        
        # 存储到JSONL
        with open(summaries_jsonl, "a") as f:
            f.write(json.dumps(summary_data) + "\n")
    
    # 5. 转换为Parquet
    summaries = []
    with open(summaries_jsonl) as f:
        for line in f:
            summaries.append(json.loads(line))
    
    pd.DataFrame(summaries).to_parquet(summaries_parquet)
    
    # 6. 向量化摘要
    embed_summaries()
```

```python
def lance_search(
    embedding: list[float],
    table: str = "message",
    limit: int = 10,
    min_sim: float = 0.0,
) -> list[tuple]:
    """
    向量搜索
    返回: [(title, content, year, month, similarity), ...]
    """
    db = get_lance_db()
    tbl = db.open_table(table)
    results = tbl.search(embedding).limit(limit).to_pandas()
    
    output = []
    for _, row in results.iterrows():
        sim = 1 / (1 + row.get("_distance", 0))
        if sim >= min_sim:
            output.append((
                row.get("conversation_title", "Untitled"),
                row.get("content", ""),
                row.get("year", 0),
                row.get("month", 0),
                sim,
            ))
    return output

def parse_json_field(value) -> list:
    """解析JSON字段（处理各种格式）"""
    if not value:
        return []
    if isinstance(value, list):
        return value
    if isinstance(value, str):
        try:
            parsed = json.loads(value)
            return parsed if isinstance(parsed, list) else []
        except json.JSONDecodeError:
            return []
    return []

def sanitize_sql_value(value: str) -> str:
    """清理SQL值（防止注入）"""
    return value.replace("'", "''")
```

## 总结

brain-mcp-cli的核心模块设计体现了以下特点：

1. **高度模块化**：每个系统职责清晰，边界明确
2. **插件化架构**：通过注册机制支持扩展
3. **抽象与实现分离**：基类定义接口，具体实现可替换
4. **单例与缓存**：优化性能，避免重复加载
5. **优雅降级**：摘要不可用时回退到原始数据
6. **配置驱动**：所有行为通过配置文件控制

这些设计使得系统既强大又灵活，易于维护和扩展。
