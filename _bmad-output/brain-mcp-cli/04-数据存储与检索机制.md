# brain-mcp-cli 数据存储与检索机制

## 1. 数据存储架构

### 1.1 存储层次结构

```
brain-mcp-cli/
├── data/                              # 数据目录
│   ├── all_conversations.parquet      # 对话数据（列式存储）
│   ├── brain_summaries_v6.parquet     # 摘要数据（列式存储）
│   ├── brain_summaries_v6.jsonl      # 摘要数据（流式格式）
│   ├── github_repos.parquet          # GitHub仓库数据
│   ├── github_commits.parquet        # GitHub提交数据
│   ├── markdown_files.parquet        # Markdown文档数据
│   ├── sync_state.json               # 同步状态
│   └── backups/                      # 备份目录
│
└── vectors/                           # 向量数据目录
    ├── brain.lance/                  # 消息向量数据库
    │   ├── data.lance               # LanceDB数据文件
    │   └── _metadata/               # 元数据索引
    │
    └── brain_summaries.lance/        # 摘要向量数据库
        ├── data.lance
        └── _metadata/
```

### 1.2 存储技术选型

| 存储类型 | 技术选择 | 原因 |
|---------|---------|------|
| **对话数据** | Parquet + DuckDB | 列式存储、高压缩率、分析查询优化 |
| **向量数据** | LanceDB | 无服务器、自动索引、混合搜索支持 |
| **摘要数据** | Parquet + JSONL | Parquet用于查询，JSONL用于流式追加 |
| **配置数据** | TOML/YAML | 人类可读、易于编辑 |

### 3.3 向量检索模式

#### 3.3.1 纯向量搜索

```python
def semantic_search(query: str, limit: int = 10):
    """语义相似度搜索"""
    # 1. 生成查询向量
    embedding = get_embedding(query)
    
    # 2. 向量搜索
    db = get_lance_db()
    tbl = db.open_table("message")
    results = tbl.search(embedding).limit(limit).to_pandas()
    
    # 3. 计算相似度
    output = []
    for _, row in results.iterrows():
        distance = row.get("_distance", 0)
        similarity = 1 / (1 + distance)  # 距离转相似度
        
        output.append({
            "title": row["conversation_title"],
            "content": row["content"],
            "similarity": similarity,
        })
    
    return output
```

```python
def filtered_search(query: str, domain: str, importance: str, limit: int = 10):
    """带过滤条件的向量搜索"""
    db = get_summaries_lance()
    tbl = db.open_table("summary")
    
    # 构建过滤条件
    filters = []
    if domain:
        filters.append(f"domain_primary = '{sanitize_sql_value(domain)}'")
    if importance:
        filters.append(f"importance = '{sanitize_sql_value(importance)}'")
    
    where_clause = " AND ".join(filters) if filters else None
    
    # 执行搜索
    embedding = get_embedding(query)
    search = tbl.search(embedding).limit(limit)
    
    if where_clause:
        search = search.where(where_clause)
    
    results = search.to_list()
    
    return results
```

### 3.4 向量维度管理

**维度不匹配检测**：
```python
def _check_dimension_mismatch(db, provider):
    """检查现有向量维度是否匹配"""
    try:
        if "message" not in db.table_names():
            return False, None
        
        tbl = db.open_table("message")
        if tbl.count_rows() == 0:
            return False, None
        
        # 采样一行获取维度
        sample = tbl.to_pandas().head(1)
        if "embedding" in sample.columns:
            existing_emb = sample["embedding"].iloc[0]
            existing_dim = len(existing_emb)
            
            if existing_dim != provider.dimension:
                return True, existing_dim
    except Exception:
        pass
    
    return False, None
```

```python
def incremental_ingest(source_path: str, state: dict) -> tuple[list[dict], dict]:
    """增量摄入（仅处理新文件）"""
    # 1. 获取上次同步时间
    last_sync = state.get("last_full_ingest")
    
    # 2. 扫描新文件
    all_files = list(Path(source_path).glob("**/*.jsonl"))
    
    if last_sync:
        new_files = [
            f for f in all_files
            if datetime.fromtimestamp(f.stat().st_mtime) > datetime.fromisoformat(last_sync)
        ]
    else:
        new_files = all_files
    
    # 3. 摄入新文件
    records = []
    for filepath in new_files:
        records.extend(parse_jsonl_file(filepath))
    
    # 4. 更新状态
    new_state = {
        "last_full_ingest": datetime.now().isoformat(),
        "last_file_count": len(all_files),
        "last_message_count": len(records),
    }
    
    return records, new_state
```

### 5.1 备份策略

```python
def create_backup():
    """创建数据备份"""
    cfg = get_config()
    backup_dir = cfg.backup_dir
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    
    backup_path = backup_dir / f"backup_{timestamp}"
    backup_path.mkdir(parents=True, exist_ok=True)
    
    # 备份Parquet文件
    shutil.copy2(cfg.parquet_path, backup_path / "conversations.parquet")
    
    if cfg.summaries_parquet.exists():
        shutil.copy2(cfg.summaries_parquet, backup_path / "summaries.parquet")
    
    # 备份LanceDB
    if cfg.lance_path.exists():
        shutil.copytree(cfg.lance_path, backup_path / "brain.lance")
    
    # 备份配置
    shutil.copy2(config_path, backup_path / "config.toml")
    
    print(f"Backup created: {backup_path}")
```

```python
    """原子写入Parquet文件"""
```

```python
def validate_record(record: dict) -> bool:
    """验证记录格式"""
    required_fields = [
        "conversation_id", "message_id", "role", "content", "created"
    ]
    
    for field in required_fields:
        if field not in record or not record[field]:
            return False
    
    if record["role"] not in ("user", "assistant"):
        return False
    
    if len(record["content"]) < 5:
        return False
    
    return True
```

```python
def validate_embedding(embedding: list[float], expected_dim: int) -> bool:
    """验证向量格式"""
    if not isinstance(embedding, list):
        return False
    
    if len(embedding) != expected_dim:
        return False
    
    if not all(isinstance(x, (int, float)) for x in embedding):

    return True
```

```python
        "embedded": 0,
        "summarized": 0,
        "sources": {},
        "domains": {},
    }
    
    # 对话统计
    if cfg.parquet_path.exists():
        df = pd.read_parquet(cfg.parquet_path)
        stats["messages"] = len(df)
        stats["conversations"] = df["conversation_id"].nunique()
        
        # 按来源统计
        stats["sources"] = df.groupby("source").size().to_dict()
    
    # 向量统计
    if cfg.lance_path.exists():
        db = lancedb.connect(str(cfg.lance_path))
        stats["embedded"] = db.open_table("message").count_rows()
    
    # 摘要统计
    if cfg.summaries_parquet.exists():
        df = pd.read_parquet(cfg.summaries_parquet)
        stats["summarized"] = len(df)
        
        # 按领域统计
        stats["domains"] = df.groupby("domain_primary").size().to_dict()
    
    return stats
```

### 8.2 健康检查

```python
def health_check() -> list[str]:
    """系统健康检查"""
    issues = []
    cfg = get_config()
    
    # 检查数据文件
    if not cfg.parquet_path.exists():
        issues.append("❌ 对话数据不存在，请运行 ingest")
    
    if not cfg.lance_path.exists():
        issues.append("⚠️  向量数据库不存在，请运行 embed")
    
    if not cfg.summaries_parquet.exists():
        issues.append("ℹ️  摘要数据不存在，高级功能受限")
    
    # 检查嵌入模型
    try:
        from brain_mcp.embed.provider import get_provider
        provider = get_provider()
        issues.append(f"✅ 嵌入模型: {provider.provider_name}")
    except ImportError:
        issues.append("❌ 嵌入模型未安装")
    
    # 检查数据一致性
    if cfg.parquet_path.exists() and cfg.lance_path.exists():
        df = pd.read_parquet(cfg.parquet_path)
        user_count = len(df[df["role"] == "user"])
        
        db = lancedb.connect(str(cfg.lance_path))
        embedded_count = db.open_table("message").count_rows()
        
        if embedded_count < user_count * 0.9:
            issues.append(f"⚠️  嵌入覆盖率低: {embedded_count}/{user_count}")
    
    return issues
```

## 总结

brain-mcp-cli的数据存储与检索机制具有以下特点：

1. **双层存储架构**：Parquet用于结构化查询，LanceDB用于向量检索
2. **增量同步机制**：支持增量摄入和嵌入，避免全量重建
3. **混合搜索能力**：支持向量搜索、全文搜索和混合搜索
4. **性能优化策略**：列式存储、批量处理、延迟加载
5. **数据一致性保证**：原子写入、数据验证
6. **监控诊断工具**：统计信息、健康检查

这些设计使得系统能够高效处理大规模对话数据，同时保持良好的查询性能和数据一致性。
