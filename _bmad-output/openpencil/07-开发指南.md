# OpenPencil 开发指南

## 快速开始

### 环境要求

- **Node.js**: >= 18.0.0
- **pnpm**: 10.24.0 (推荐使用 pnpm)
- **操作系统**: Windows / macOS / Linux

### 安装依赖

```bash
# 克隆项目
git clone <repository-url>
cd OpenPencil

# 安装依赖
pnpm install
```

### 运行项目

```bash
# 运行桌面应用
cd apps/desktop
pnpm dev

# 运行核心库开发
cd packages/core
pnpm dev
```

### 运行测试

```bash
cd packages/core

# 运行测试
pnpm test

# 运行测试 (UI 界面)
pnpm test:ui

# 运行测试并生成覆盖率
pnpm test:coverage
```

### 构建项目

```bash
# 构建核心库
cd packages/core
pnpm build

# 构建桌面应用
cd apps/desktop
pnpm build
```

#### 2. 添加验证逻辑

```typescript
// packages/core/src/format/validator.ts
function validateShape(shape: any, layerIndex: number): string[] {
  const errors: string[] = [];
  const prefix = `layers[${layerIndex}].shape`;
  
  switch (shape.type) {
    // ... 其他图形
    
    case 'triangle':
      if (typeof shape.x1 !== 'number') errors.push(`${prefix}.x1 must be a number`);
      if (typeof shape.y1 !== 'number') errors.push(`${prefix}.y1 must be a number`);
      if (typeof shape.x2 !== 'number') errors.push(`${prefix}.x2 must be a number`);
      if (typeof shape.y2 !== 'number') errors.push(`${prefix}.y2 must be a number`);
      if (typeof shape.x3 !== 'number') errors.push(`${prefix}.x3 must be a number`);
      if (typeof shape.y3 !== 'number') errors.push(`${prefix}.y3 must be a number`);
      if (typeof shape.fill !== 'string') errors.push(`${prefix}.fill must be a string`);
      break;
  }
  
  return errors;
}
```

#### 3. 创建渲染器

```typescript
// packages/core/src/renderer/shapes/triangle.ts
import type { TriangleShape } from '../../types';

export function renderTriangle(
  ctx: CanvasRenderingContext2D,
  shape: TriangleShape
): void {
  ctx.beginPath();
  ctx.moveTo(shape.x1, shape.y1);
  ctx.lineTo(shape.x2, shape.y2);
  ctx.lineTo(shape.x3, shape.y3);
  ctx.closePath();
  
  if (shape.fill) {
    ctx.fillStyle = shape.fill;
    ctx.fill();
  }
  
  if (shape.stroke && shape.strokeWidth) {
    ctx.strokeStyle = shape.stroke;
    ctx.lineWidth = shape.strokeWidth;
    ctx.stroke();
  }
}
```

#### 5. 集成到主渲染器

```typescript
// packages/core/src/renderer/canvas-renderer.ts
import { renderTriangle } from './shapes/triangle';

export class CanvasRenderer {
  private renderLayer(layer: any): void {
    this.ctx.save();
    this.ctx.globalAlpha = layer.opacity;
    
    switch (layer.shape.type) {
      // ... 其他图形
      case 'triangle':
        renderTriangle(this.ctx, layer.shape);
        break;
    }
    
    this.ctx.restore();
  }
}
```
