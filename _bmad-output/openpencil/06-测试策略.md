# OpenPencil 测试策略

## 概述

OpenPencil 采用测试驱动开发 (TDD) 方法,使用 Vitest 作为测试框架,配合 @testing-library/react 进行组件测试。项目当前包含 36 个测试用例,覆盖类型验证、格式解析、图形渲染和 UI 组件等多个层面。

## 测试用例详解

### 1. 格式解析测试 (format.test.ts)

#### 测试内容
- 有效文档解析
- 无效文档错误处理
- 文档序列化

#### 示例代码

```typescript
import { describe, it, expect } from 'vitest';
import { PencilParser } from '../src/format';

describe('PencilParser', () => {
  it('should parse valid document', () => {
    const json = JSON.stringify({
      version: '1.0.0',
      metadata: {
        created: '2026-02-21T10:00:00Z',
        modified: '2026-02-21T10:00:00Z',
        title: 'Test'
      },
      canvas: { width: 800, height: 600, backgroundColor: '#FFFFFF' },
      layers: []
    });
    
    const parser = new PencilParser();
    const doc = parser.parse(json);
    
    expect(doc.version).toBe('1.0.0');
    expect(doc.canvas.width).toBe(800);
  });
  
  it('should throw on invalid document', () => {
    const parser = new PencilParser();
    expect(() => parser.parse('{}')).toThrow('Invalid document');
  });
  
  it('should serialize document', () => {
    const parser = new PencilParser();
    const doc: any = { version: '1.0.0', metadata: {}, canvas: {}, layers: [] };
    const json = parser.serialize(doc);
    expect(JSON.parse(json)).toEqual(doc);
  });
});
```

### 4. 组件测试 (LayerPanel.test.tsx)

#### 测试内容

- 重命名功能

#### 示例代码

```typescript
import { describe, it, expect, vi } from 'vitest';
import { render, screen } from '@testing-library/react';
import { userEvent } from '@testing-library/user-event';
import { LayerPanel } from '../../src/components/LayerPanel';
import type { PencilDocument } from '../../src/types';

describe('LayerPanel', () => {
  const mockDoc: PencilDocument = {
    version: '1.0.0',
    metadata: { created: '', modified: '', title: 'Test' },
    canvas: { width: 800, height: 600, backgroundColor: '#FFFFFF' },
    layers: [
      {
        id: 'layer-1',
        type: 'shape',
        name: 'Rectangle Layer',
        visible: true,
        locked: false,
        opacity: 1,
        shape: {
          type: 'rectangle',
          x: 10, y: 10,
          width: 100, height: 50,
          fill: '#FF0000'
        }
      },
      {
        id: 'layer-2',
        type: 'shape',
        name: 'Circle Layer',
        visible: false,
        locked: false,
        opacity: 0.8,
        shape: {
          type: 'circle',
          cx: 100,
          cy: 100,
          radius: 50,
          fill: '#00FF00'
        }
      }
    ]
  };
  
  it('should render layer list', () => {
    render(
      <LayerPanel
        document={mockDoc}
        onLayerSelect={vi.fn()}
        onLayerReorder={vi.fn()}
        onLayerToggleVisible={vi.fn()}
        onLayerDelete={vi.fn()}
        onLayerRename={vi.fn()}
        onLayerOpacityChange={vi.fn()}
      />
    );
    
    expect(screen.getByText('Rectangle Layer')).toBeInTheDocument();
    expect(screen.getByText('Circle Layer')).toBeInTheDocument();
  });
  
  it('should toggle layer visibility', async () => {
    const user = userEvent.setup();
    const onToggle = vi.fn();
    
    render(
      <LayerPanel
        document={mockDoc}
        onLayerSelect={vi.fn()}
        onLayerReorder={vi.fn()}
        onLayerToggleVisible={onToggle}
        onLayerDelete={vi.fn()}
        onLayerRename={vi.fn()}
        onLayerOpacityChange={vi.fn()}
      />
    );
    
    const visibilityButtons = screen.getAllByRole('button', { name: /toggle visibility/i });
    await user.click(visibilityButtons[0]);
    
    expect(onToggle).toHaveBeenCalledWith('layer-1');
  });
  
  it('should delete layer', async () => {
    const user = userEvent.setup();
    const onDelete = vi.fn();
    
    render(
      <LayerPanel
        document={mockDoc}
        selectedLayerId="layer-1"
        onLayerSelect={vi.fn()}
        onLayerReorder={vi.fn()}
        onLayerToggleVisible={vi.fn()}
        onLayerDelete={onDelete}
        onLayerRename={vi.fn()}
        onLayerOpacityChange={vi.fn()}
      />
    );
    
    const deleteButtons = screen.getAllByRole('button', { name: /delete layer/i });
    await user.click(deleteButtons[0]);
    
    expect(onDelete).toHaveBeenCalledWith('layer-1');
  });
});
```

## 测试最佳实践

### 1. 测试命名

使用清晰的描述性名称:

```typescript
// 好的命名
it('should throw error when radius is zero', () => { /* ... */ });

// 不好的命名
it('test1', () => { /* ... */ });
```

### 2. 测试结构

使用 AAA 模式 (Arrange, Act, Assert):

```typescript
it('should toggle layer visibility', async () => {
  // Arrange
  const user = userEvent.setup();
  const onToggle = vi.fn();
  render(<LayerPanel document={mockDoc} onLayerToggleVisible={onToggle} {...otherProps} />);
  
  // Act
  const button = screen.getByRole('button', { name: /toggle visibility/i });
  await user.click(button);
  
  // Assert
  expect(onToggle).toHaveBeenCalledWith('layer-1');
});
```

### 3. 测试隔离

每个测试应该独立运行:

```typescript
beforeEach(() => {
  // 重置测试环境
  canvas = document.createElement('canvas');
  renderer = new CanvasRenderer(canvas);
});

afterEach(() => {
  // 清理测试环境
  vi.clearAllMocks();
});
```

### 4. Mock 使用

合理使用 Mock:

```typescript
// Mock Canvas 上下文
const mockCtx = {
  fillRect: vi.fn(),
  strokeRect: vi.fn(),
  // ...
};
canvas.getContext = () => mockCtx as any;

// 验证 Mock 调用
expect(mockCtx.fillRect).toHaveBeenCalledWith(100, 100, 200, 150);
```

### 5. 边界测试

测试边界情况:

```typescript
describe('Circle radius validation', () => {
  it('should accept radius > 0', () => {
    const result = validateDocument({ /* radius: 1 */ });
    expect(result.valid).toBe(true);
  });
  
  it('should reject radius = 0', () => {
    const result = validateDocument({ /* radius: 0 */ });
    expect(result.valid).toBe(false);
  });
  
  it('should reject radius < 0', () => {
    const result = validateDocument({ /* radius: -1 */ });
    expect(result.valid).toBe(false);
  });
});
```
