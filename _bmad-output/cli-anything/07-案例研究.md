### 渲染流水线

```python
def render_project(project: dict, output_path: str, preset: str):
    """渲染项目到音频文件"""
    
    # 1. 收集所有需要渲染的轨道
    tracks_to_render = []
    for track in project["tracks"]:
        if track["solo"]:
            tracks_to_render = [track]
            break
        if not track["mute"]:
            tracks_to_render.append(track)
    
    # 2. 渲染每个轨道
    rendered_tracks = []
    for track in tracks_to_render:
        track_audio = []
        
        # 2.1 处理每个片段
        for clip in track["clips"]:
            audio = read_wav(clip["source"])
            audio = trim_audio(audio, clip["start"], clip["end"])
            audio = place_at_position(audio, clip["offset"])
            track_audio.append(audio)
        
        # 2.2 混合轨道内的片段
        mixed_track = mix_audio(track_audio)
        
        # 2.3 应用效果链
        for effect in track["effects"]:
            mixed_track = apply_effect(mixed_track, effect)
        
        # 2.4 应用轨道音量和声像
        mixed_track = apply_volume(mixed_track, track["volume"])
        mixed_track = apply_pan(mixed_track, track["pan"])
        
        rendered_tracks.append(mixed_track)
    
    # 3. 混合所有轨道
    final_audio = mix_audio(rendered_tracks)
    
    # 4. 限制到 [-1, 1] 范围
    final_audio = clamp_audio(final_audio)
    
    # 5. 写入输出文件
    write_wav(output_path, final_audio, project["settings"])
```

### bpy 脚本生成

```python
def generate_bpy_script(scene: dict) -> str:
    """生成 Blender Python 脚本"""
    script_lines = [
        "import bpy",
        "import math",
        "",
        "# 清除默认场景",
        "bpy.ops.object.select_all(action='SELECT')",
        "bpy.ops.object.delete()",
        "",
        "# 创建对象",
    ]
    
    for obj in scene.get("objects", []):
        script_lines.extend([
            f"# 创建 {obj['name']}",
            f"bpy.ops.mesh.primitive_{obj['primitive'].lower()}_add()",
            f"obj = bpy.context.active_object",
            f"obj.name = '{obj['name']}'",
            f"obj.location = {obj['location']}",
            f"obj.rotation_euler = {obj['rotation']}",
            f"obj.scale = {obj['scale']}",
            "",
        ])
        
        # 添加材质
        if obj.get("material"):
            script_lines.extend([
                f"mat = bpy.data.materials.new('{obj['material']}')",
                f"mat.diffuse_color = {obj.get('material_color', [0.8, 0.8, 0.8, 1.0])}",
                f"obj.data.materials.append(mat)",
                "",
            ])
        
        # 添加修改器
        for mod in obj.get("modifiers", []):
            script_lines.extend([
                f"mod = obj.modifiers.new('{mod['type']}', '{mod['type']}')",
            ])
            for key, value in mod.get("params", {}).items():
                script_lines.append(f"mod.{key} = {value}")
            script_lines.append("")
    
    # 添加相机
    for cam in scene.get("cameras", []):
        script_lines.extend([
            f"bpy.ops.object.camera_add(location={cam['location']})",
            f"cam_obj = bpy.context.active_object",
            f"cam_obj.name = '{cam['name']}'",
            f"cam_obj.rotation_euler = {cam['rotation']}",
            f"cam_obj.data.lens = {cam['lens']}",
            "",
        ])
    
    # 添加灯光
    for light in scene.get("lights", []):
        script_lines.extend([
            f"bpy.ops.object.light_add(type='{light['type']}', location={light['location']})",
            f"light_obj = bpy.context.active_object",
            f"light_obj.name = '{light['name']}'",
            f"light_obj.data.energy = {light['energy']}",
            "",
        ])
    
    # 设置渲染参数
    settings = scene.get("settings", {})
    script_lines.extend([
        "# 渲染设置",
        f"bpy.context.scene.render.fps = {settings.get('fps', 24)}",
        f"bpy.context.scene.frame_start = {settings.get('frame_start', 1)}",
        f"bpy.context.scene.frame_end = {settings.get('frame_end', 250)}",
        f"bpy.context.scene.render.resolution_x = {settings.get('resolution_x', 1920)}",
        f"bpy.context.scene.render.resolution_y = {settings.get('resolution_y', 1080)}",
    ])
    
    return "\n".join(script_lines)
```

### Backend 执行

```python
def render_scene(scene_path: str, output_path: str, engine: str = "CYCLES"):
    """渲染 3D 场景"""
    
    # 1. 加载场景
    with open(scene_path) as f:
        scene = json.load(f)
    
    # 2. 生成 bpy 脚本
    script = generate_bpy_script(scene)
    script_path = scene_path.replace(".json", "_script.py")
    with open(script_path, "w") as f:
        f.write(script)
    
    # 3. 查找 Blender
    blender_path = find_blender()
    
    # 4. 执行渲染
    subprocess.run([
        blender_path,
        "--background",
        "--python", script_path,
        "--render-output", output_path,
        "--render-frame", "1..250",
        "--engine", engine
    ], check=True)
    
    return {"output": output_path, "engine": engine}
```

def render_project(project: dict, output_path: str, format: str = "png"):
    """渲染 GIMP 项目"""
    
    # 1. 创建输出图像
    output = Image.new(
        "RGBA",
        (project["width"], project["height"]),
        (0, 0, 0, 0)
    )
    
    # 2. 从下到上处理图层
    for layer in reversed(project["layers"]):
        if not layer["visible"]:
            continue
        
        # 2.1 渲染图层内容
        layer_image = render_layer_content(layer)
        
        # 2.2 应用滤镜
        for filter_def in layer.get("filters", []):
            layer_image = apply_filter(layer_image, filter_def)
        
        # 2.3 应用混合模式
        output = blend_layers(
            output,
            layer_image,
            layer["blend_mode"],
            layer["opacity"] / 100
        )
    
    # 3. 保存输出
    output.save(output_path, format.upper())
