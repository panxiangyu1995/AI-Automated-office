### 1. Plugin Manager

```python
"""
插件管理器 - 管理 CLI-Anything 插件的注册、发现和调用
"""

from dataclasses import dataclass
from typing import Dict, List, Optional
import subprocess
import json

@dataclass
class PluginInfo:
    """插件信息"""
    name: str
    cli_command: str
    skill_file: str
    description: str
    version: str
    capabilities: List[str]

class PluginManager:
    """插件管理器"""
    
    def __init__(self):
        self.plugins: Dict[str, PluginInfo] = {}
        self.skill_cache: Dict[str, dict] = {}
    
    def register_plugin(self, plugin_config: dict) -> None:
        """注册插件"""
        plugin = PluginInfo(
            name=plugin_config["name"],
            cli_command=plugin_config["cli_command"],
            skill_file=plugin_config.get("skill_file", "skills/SKILL.md"),
            description=plugin_config.get("description", ""),
            version=plugin_config.get("version", "1.0.0"),
            capabilities=plugin_config.get("capabilities", [])
        )
        self.plugins[plugin.name] = plugin
    
    def discover_plugins(self, plugin_dir: str) -> List[PluginInfo]:
        """发现目录中的所有插件"""
        import os
        from pathlib import Path
        
        discovered = []
        plugin_path = Path(plugin_dir)
        
        for plugin_folder in plugin_path.iterdir():
            if plugin_folder.is_dir():
                manifest_file = plugin_folder / "plugin.json"
                if manifest_file.exists():
                    with open(manifest_file) as f:
                        config = json.load(f)
                    self.register_plugin(config)
                    discovered.append(self.plugins[config["name"]])
        
        return discovered
    
    def get_plugin(self, name: str) -> Optional[PluginInfo]:
        """获取插件信息"""
        return self.plugins.get(name)
    
    def list_plugins(self) -> List[PluginInfo]:
        """列出所有插件"""
        return list(self.plugins.values())
    
    def find_plugin_by_capability(self, capability: str) -> List[PluginInfo]:
        """根据能力查找插件"""
        return [
            p for p in self.plugins.values()
            if capability in p.capabilities
        ]
```

```python
    """Skill 定义"""

    software: str

    prerequisites: List[str]

    """Skill 注册表"""
    
    def __init__(self):
        self.skills: Dict[str, Skill] = {}
    
    def load_skill(self, plugin: PluginInfo) -> Skill:
        """加载插件的 Skill"""
        if plugin.name in self.skills:
            return self.skills[plugin.name]
        
        skill_path = self._find_skill_file(plugin)
        skill = self._parse_skill(skill_path)
        self.skills[plugin.name] = skill
        return skill
    
    def _find_skill_file(self, plugin: PluginInfo) -> Path:
        """查找 Skill 文件"""
        # 尝试多个位置
        locations = [
            Path(plugin.skill_file),
            Path(f"plugins/{plugin.name}/{plugin.skill_file}"),
            Path(f"~/.cli-anything-{plugin.name}/skills/SKILL.md").expanduser(),
        ]
        
        for loc in locations:
            if loc.exists():
                return loc
        
        raise FileNotFoundError(f"Skill file not found for plugin: {plugin.name}")
```

### 3. Command Generator

```python
"""
命令生成器 - 根据意图生成 CLI 命令
"""

from typing import Dict, List, Optional
import shlex

class CommandGenerator:
    """命令生成器"""
    
    def __init__(self, plugin_manager: PluginManager, skill_registry: SkillRegistry):
        self.plugin_manager = plugin_manager
        self.skill_registry = skill_registry
    
    def generate_command(
        self,
        plugin_name: str,
        command_path: str,
        params: Dict[str, any],
        json_output: bool = True
    ) -> str:
        """生成 CLI 命令"""
        plugin = self.plugin_manager.get_plugin(plugin_name)
        if not plugin:
            raise ValueError(f"Plugin not found: {plugin_name}")
        
        # 构建命令
        parts = [plugin.cli_command]
        
        if json_output:
            parts.append("--json")
        
        # 添加命令路径
        parts.extend(command_path.split())
        
        # 添加参数
        for key, value in params.items():
            if isinstance(value, bool):
                if value:
                    parts.append(f"--{key}")
            elif isinstance(value, list):
                for v in value:
                    parts.extend([f"--{key}", str(v)])
            else:
                parts.extend([f"--{key}", str(value)])
        
        return " ".join(shlex.quote(p) if " " in p else p for p in parts)
    
    def generate_command_sequence(
        self,
        plugin_name: str,
        operations: List[Dict[str, any]]
    ) -> List[str]:
        """生成命令序列"""
        commands = []
        for op in operations:
            cmd = self.generate_command(
                plugin_name=plugin_name,
                command_path=op["command"],
                params=op.get("params", {}),
                json_output=op.get("json_output", True)
            )
            commands.append(cmd)
        return commands
    
    def generate_from_template(
        self,
        template_name: str,
        params: Dict[str, any]
    ) -> List[str]:
        """从模板生成命令序列"""
        templates = self._load_templates()
        if template_name not in templates:
            raise ValueError(f"Template not found: {template_name}")
        
        template = templates[template_name]
        return [
            cmd.format(**params)
            for cmd in template["commands"]
        ]
    
    def _load_templates(self) -> Dict[str, dict]:
        """加载命令模板"""
        return {
            "blur_image": {
                "description": "Apply blur filter to image",
                "commands": [
                    "cli-anything-gimp project new -o {temp_project}",
                    "cli-anything-gimp --project {temp_project} layer import 0 {input_file}",
                    "cli-anything-gimp --project {temp_project} filter apply blur --radius {radius}",
                    "cli-anything-gimp --project {temp_project} export render {output_file} --overwrite"
                ]
            },
            "normalize_audio": {
                "description": "Normalize audio levels",
                "commands": [
                    "cli-anything-audacity project new -o {temp_project}",
                    "cli-anything-audacity --project {temp_project} track add --name main",
                    "cli-anything-audacity --project {temp_project} clip add 0 {input_file}",
                    "cli-anything-audacity --project {temp_project} effect add normalize --track 0 -p target_db={target_db}",
                    "cli-anything-audacity --project {temp_project} export render {output_file} --overwrite"
                ]
            }
        }
```

### 4. Context Manager

```python
"""
上下文管理器 - 管理 Agent 的上下文和 Token 预算
"""

from typing import Dict, List, Optional
import json

@dataclass
class ExecutionContext:
    """执行上下文"""
    session_id: str
    active_project: Optional[str]
    loaded_skills: List[str]
    token_budget: int
    tokens_used: int

class ContextManager:
    """上下文管理器"""
    
    def __init__(self, max_token_budget: int = 4000):
        self.max_token_budget = max_token_budget
        self.contexts: Dict[str, ExecutionContext] = {}
    
    def create_context(self, session_id: str) -> ExecutionContext:
        """创建新的执行上下文"""
        context = ExecutionContext(
            session_id=session_id,
            active_project=None,
            loaded_skills=[],
            token_budget=self.max_token_budget,
            tokens_used=0
        )
        self.contexts[session_id] = context
        return context
    
    def get_context(self, session_id: str) -> Optional[ExecutionContext]:
        """获取执行上下文"""
        return self.contexts.get(session_id)
    
    def load_skill_to_context(
        self,
        session_id: str,
        skill: Skill
    ) -> int:
        """加载 Skill 到上下文，返回 Token 成本"""
        context = self.get_context(session_id)
        if not context:
            raise ValueError(f"Context not found: {session_id}")
        
        # 估算 Skill 的 Token 成本
        skill_tokens = self._estimate_skill_tokens(skill)
        
        if context.tokens_used + skill_tokens > context.token_budget:
            raise TokenBudgetExceeded(
                f"Loading skill would exceed token budget. "
                f"Current: {context.tokens_used}, Skill: {skill_tokens}, Budget: {context.token_budget}"
            )
        
        context.loaded_skills.append(skill.name)
        context.tokens_used += skill_tokens
        
        return skill_tokens
    
    def execute_with_budget(
        self,
        session_id: str,
        command: str,
        estimated_tokens: int
    ) -> dict:
        """在预算内执行命令"""
        context = self.get_context(session_id)
        if not context:
            raise ValueError(f"Context not found: {session_id}")
        
        if context.tokens_used + estimated_tokens > context.token_budget:
            raise TokenBudgetExceeded()
        
        # 执行命令
        result = subprocess.run(
            command.split(),
            capture_output=True,
            text=True
        )
        
        # 计算实际 Token
        actual_tokens = self._count_tokens(result.stdout)
        context.tokens_used += actual_tokens
        
        return {
            "stdout": result.stdout,
            "stderr": result.stderr,
            "return_code": result.returncode,
            "tokens_used": actual_tokens,
            "remaining_budget": context.token_budget - context.tokens_used
        }
    
    def _estimate_skill_tokens(self, skill: Skill) -> int:
        """估算 Skill 的 Token 数量"""
        # 简化估算：基于命令数量
        base = 50  # 基础开销
        command_cost = sum(
            len(group.commands) * 10
            for group in skill.command_groups
        )
        return base + command_cost
    
    def _count_tokens(self, text: str) -> int:
        """计算文本的 Token 数量"""
        # 简化计算：每 4 个字符约 1 个 token
        return len(text) // 4
```

### 5. Plugin Executor

```python
"""
插件执行器 - 执行 CLI 命令并处理结果
"""

import subprocess
import json
from typing import Dict, List, Optional

class PluginExecutor:
    """插件执行器"""
    
    def __init__(
        self,
        plugin_manager: PluginManager,
        context_manager: ContextManager
    ):
        self.plugin_manager = plugin_manager
        self.context_manager = context_manager
    
    def execute_command(
        self,
        session_id: str,
        command: str,
        timeout: int = 30
    ) -> dict:
        """执行单个命令"""
        try:
            result = subprocess.run(
                command.split(),
                capture_output=True,
                text=True,
                timeout=timeout
            )
            
            # 解析 JSON 输出
            output = result.stdout
            if output.strip().startswith("{"):
                try:
                    output = json.loads(output)
                except json.JSONDecodeError:
                    pass
            
            return {
                "success": result.returncode == 0,
                "output": output,
                "error": result.stderr if result.returncode != 0 else None,
                "return_code": result.returncode
            }
        
        except subprocess.TimeoutExpired:
            return {
                "success": False,
                "output": None,
                "error": f"Command timed out after {timeout} seconds",
                "return_code": -1
            }
    
    def execute_sequence(
        self,
        session_id: str,
        commands: List[str],
        stop_on_error: bool = True
    ) -> List[dict]:
        """执行命令序列"""
        results = []
        
        for cmd in commands:
            result = self.execute_command(session_id, cmd)
            results.append(result)
            
            if stop_on_error and not result["success"]:
                break
        
        return results
    
    def execute_workflow(
        self,
        session_id: str,
        workflow: Dict[str, any]
    ) -> dict:
        """执行工作流"""
        plugin_name = workflow["plugin"]
        operations = workflow["operations"]
        
        # 生成命令序列
        generator = CommandGenerator(
            self.plugin_manager,
            SkillRegistry()
        )
        commands = generator.generate_command_sequence(plugin_name, operations)
        
        # 执行命令
        results = self.execute_sequence(session_id, commands)
        
        # 汇总结果
        return {
            "workflow": workflow.get("name", "unnamed"),
            "plugin": plugin_name,
            "total_commands": len(commands),
            "successful": sum(1 for r in results if r["success"]),
            "failed": sum(1 for r in results if not r["success"]),
            "results": results
        }
```

```python
"""

    """Agent 插件集成示例"""
    
    def __init__(self):
        # 初始化组件
        self.plugin_manager = PluginManager()
        self.skill_registry = SkillRegistry()
        self.context_manager = ContextManager()
        self.executor = PluginExecutor(
            self.plugin_manager,
            self.context_manager
        )
        
        # 发现并注册插件
        self.plugin_manager.discover_plugins("plugins/")
    
    def process_user_request(self, user_request: str) -> dict:
        """处理用户请求"""
        # 1. 创建执行上下文
        session_id = self._generate_session_id()
        context = self.context_manager.create_context(session_id)
        
        # 2. 分析请求，确定需要的插件
        plugin = self._match_plugin(user_request)
        if not plugin:
            return {"error": "No suitable plugin found"}
        
        # 3. 加载 Skill
        skill = self.skill_registry.load_skill(plugin)
        token_cost = self.context_manager.load_skill_to_context(
            session_id, skill
        )
        
        # 4. 生成命令
        generator = CommandGenerator(
            self.plugin_manager,
            self.skill_registry
        )
        
        # 5. 执行命令
        workflow = self._create_workflow(plugin, user_request)
        result = self.executor.execute_workflow(session_id, workflow)
        
        return {
            "session_id": session_id,
```

```yaml
  format: "%(asctime)s - %(name)s - %(levelname)s - %(message)s"
```

## 总结

集成 CLI-Anything 到 AI-Automated-office 的关键步骤：

1. **插件注册**：通过 PluginManager 管理插件生命周期
2. **Skill 加载**：通过 SkillRegistry 解析和缓存 Skill 定义
3. **命令生成**：通过 CommandGenerator 生成 CLI 命令
4. **上下文管理**：通过 ContextManager 管理 Token 预算
5. **命令执行**：通过 PluginExecutor 执行命令并处理结果

这套架构提供了：
- **标准化接口**：统一的插件调用方式
- **Token 优化**：智能的上下文管理
- **可扩展性**：易于添加新插件
- **可靠性**：完善的错误处理和重试机制
